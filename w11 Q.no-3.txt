
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        Map<Integer, Integer> count = new HashMap<>();
        Map<Integer, Integer> first = new HashMap<>();
        int max = 0, len = n;

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            first.putIfAbsent(a[i], i);
            int c = count.getOrDefault(a[i], 0) + 1;
            count.put(a[i], c);
            if (c > max) {
                max = c;
                len = i - first.get(a[i]) + 1;
            } else if (c == max) {
                len = Math.min(len, i - first.get(a[i]) + 1);
            }
        }
        System.out.println(len);
    }
}

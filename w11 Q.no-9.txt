
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] count = new int[6];

        for (int i = 0; i < n; i++)
            count[sc.nextInt()]++;

        int max = 0, id = 0;
        for (int i = 1; i <= 5; i++) {
            if (count[i] > max) {
                max = count[i];
                id = i;
            }
        }
        System.out.println(id);
    }
}


import java.util.*;

class Product {
    int id, stock;
    String name;
    double price;

    Product(int id, String name, int stock, double price) {
        this.id = id;
        this.name = name;
        this.stock = stock;
        this.price = price;
    }

    public String toString() {
        return id + " " + name + " " + stock + " " + price;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        Map<Integer, Product> map = new HashMap<>();

        for (int i = 0; i < n; i++) {
            int id = sc.nextInt();
            String name = sc.next();
            int stock = sc.nextInt();
            double price = sc.nextDouble();
            map.put(id, new Product(id, name, stock, price));
        }

        int id = sc.nextInt();
        int sold = sc.nextInt();

        if (map.containsKey(id))
            map.get(id).stock = Math.max(0, map.get(id).stock - sold);

        int threshold = sc.nextInt();
        System.out.println("Low Stock:");
        for (Product p : map.values())
            if (p.stock < threshold)
                System.out.println(p);

        List<Product> list = new ArrayList<>(map.values());
        list.sort(Comparator.comparingInt(p -> p.stock));

        System.out.println("Sorted by Stock:");
        for (Product p : list)
            System.out.println(p);

        System.out.println("Search Product ID:");
        int searchId = sc.nextInt();
        if (map.containsKey(searchId))
            System.out.println(map.get(searchId));
        else
            System.out.println("Product not found");

        sc.close();
    }
}

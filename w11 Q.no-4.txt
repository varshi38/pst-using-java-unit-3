
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<List<Object>> list = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            String name = sc.next();
            double grade = sc.nextDouble();
            list.add(Arrays.asList(name, grade));
        }

        TreeSet<Double> grades = new TreeSet<>();
        for (List<Object> s : list)
            grades.add((Double) s.get(1));

        double second = grades.toArray(new Double[0])[1];
        List<String> names = new ArrayList<>();

        for (List<Object> s : list)
            if ((Double) s.get(1) == second)
                names.add((String) s.get(0));

        Collections.sort(names);
        for (String name : names)
            System.out.println(name);
    }
}
